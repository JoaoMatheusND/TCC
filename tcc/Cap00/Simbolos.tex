\chapter*{Lista de símbolos}

\begin{description}
  \item[$f_s$] \quad Taxa de amostragem (amostras por segundo)
  \item[$f_{clk}$] \quad Frequência do clock do sistema
  \item[$f_{sys}$] \quad Frequência do clock do sistema fornecida à máquina de estados do PIO
  \item[$f_{sig}$] \quad Frequência do sinal de referência aplicado à entrada
  \item[$T_s$] \quad Período de amostragem, $1/f_s$
  \item[$C_{iter}$] \quad Custo do laço de amostragem, em ciclos de clock por amostra
  \item[$P$] \quad Número de amostras por período do sinal de referência, $f_s/f_{sig}$
  \item[$\varphi$] \quad Parte fracionária de $P$
  \item[$\tau_k$] \quad Intervalo medido entre a $k$-ésima borda e a seguinte
  \item[$\sigma$] \quad Desvio-padrão dos intervalos entre bordas
  \item[$\sigma_q$] \quad Desvio-padrão atribuível à quantização do amostrador ideal
  \item[$\sigma_{exc}$] \quad Excedente de jitter sobre o limite de quantização
  \item[$\eta$] \quad Fração de amostras entregues em captura contínua
  \item[$O$] \quad Ocupação de CPU durante a captura
  \item[$\Delta n$] \quad Diferença entre a contagem de bordas do DAQ e a da referência
  \item[$r$] \quad Tamanho de efeito do teste de Mann-Whitney
  \item[$p_{\mathrm{Holm}}$] \quad P-valor ajustado pela correção de Holm-Bonferroni
  \item[$N$] \quad Número de repetições por ponto de medição
\end{description}
